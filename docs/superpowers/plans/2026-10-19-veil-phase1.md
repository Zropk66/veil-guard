# veil 阶段一实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 交付 Windows x64 客户端防护库的第一阶段：密码学基座、syscall 平台层、四个观测 provider、证据派生链，以及可选的授权信封层。执行重映射留待阶段二。

**Architecture:** header-only 分层库，L0 密码学 → L1 平台 → L2 provider → L3 证据派生 → L4 可选授权。provider 只输出正常环境恒为零的原始观测字节，判定被彻底移除，篡改表现为 HKDF 派生分叉。可选 PIMPL 静态库 target 与 header-only 共用同一套源。

**Tech Stack:** C++20、CMake 3.20+、Ninja、MinGW-w64 g++ 与 MSVC 双支持、vendored Monocypher 4.0.3、Node.js 24 用于跨实现夹具。

## 阶段范围

本计划只覆盖阶段一。以下明确排除，留待阶段二：

- 执行重映射（私有可执行页、重定位修正、`.pdata` 展开信息登记）
- `PAGE_GUARD` 重定向探测
- VEH/SEH 投递分析
- TLS callback 入口点保护
- PIMPL 静态库 target（阶段一先留出 `VEIL_HEADER_ONLY` 宏与目录结构，实现放阶段二）

阶段一交付后必须满足：全部观测结构在未篡改环境下逐字节为零；篡改任一字节导致派生分叉；`ctest` 与跨实现夹具测试全绿。

```cpp
#pragma once

#include <cstddef>
#include <cstdint>

namespace veil::detail {

// volatile 写入阻止编译器把擦除优化掉。
VEIL_NOINLINE inline void secure_wipe(void* buffer, std::size_t size) noexcept {
    volatile unsigned char* p = static_cast<volatile unsigned char*>(buffer);
    while (size-- > 0)
        *p++ = 0;
}

// 返回 1 表示相等。调用方不得把结果直接当作分支条件使用。
VEIL_FORCEINLINE int constant_time_compare(const void* a, const void* b,
                                           std::size_t size) noexcept {
    const auto* pa = static_cast<const unsigned char*>(a);
    const auto* pb = static_cast<const unsigned char*>(b);
    unsigned char diff = 0;
    for (std::size_t i = 0; i < size; ++i)
        diff |= static_cast<unsigned char>(pa[i] ^ pb[i]);
    return diff == 0 ? 1 : 0;
}

VEIL_FORCEINLINE void store_le64(std::uint8_t* out, std::uint64_t value) noexcept {
    for (std::size_t i = 0; i < 8; ++i)
        out[i] = static_cast<std::uint8_t>(value >> (i * 8));
}

VEIL_FORCEINLINE void store_le32(std::uint8_t* out, std::uint32_t value) noexcept {
    for (std::size_t i = 0; i < 4; ++i)
        out[i] = static_cast<std::uint8_t>(value >> (i * 8));
}

}  // namespace veil::detail
```

```cpp
#pragma once

#include "../config.hpp"

#include <cstddef>
#include <cstdint>

namespace veil::detail {

VEIL_FORCEINLINE constexpr std::uint64_t mix64(std::uint64_t x) noexcept {
    x ^= x >> 33;
    x *= 0xFF51AFD7ED558CCDULL;
    x ^= x >> 33;
    x *= 0xC4CEB9FE1A85EC53ULL;
    x ^= x >> 33;
    return x;
}

VEIL_FORCEINLINE constexpr std::uint32_t fnv1a32(const char* text, std::uint32_t salt) noexcept {
    std::uint32_t h = 0x811C9DC5u ^ salt;
    while (*text != '\0') {
        h ^= static_cast<std::uint32_t>(static_cast<unsigned char>(*text++));
        h *= 0x01000193u;
    }
    return h;
}

VEIL_FORCEINLINE constexpr std::uint32_t crc32_step(std::uint32_t crc, unsigned char byte) noexcept {
    crc ^= byte;
    for (int bit = 0; bit < 8; ++bit)
        crc = (crc >> 1) ^ (0xEDB88320u & (0u - (crc & 1u)));
    return crc;
}

VEIL_FORCEINLINE std::uint32_t crc32(const void* data, std::size_t size) noexcept {
    const auto* p = static_cast<const unsigned char*>(data);
    std::uint32_t crc = 0xFFFFFFFFu;
    for (std::size_t i = 0; i < size; ++i)
        crc = crc32_step(crc, p[i]);
    return ~crc;
}

// 以槽位序号派生独立盐，避免不同调用点共享常量。
template <std::uint64_t Slot>
struct Salt {
    static constexpr std::uint64_t value =
        mix64(BUILD_SEED ^ (Slot * 0x2545F4914F6CDD1DULL));
    static constexpr std::uint32_t hash_salt =
        static_cast<std::uint32_t>(value ^ (value >> 32));
};

}  // namespace veil::detail
```

```cpp
#pragma once

#include "../config.hpp"

#include <concepts>

namespace veil {

struct Observation {
    const void* data;
    std::size_t size;
};

// provider 只输出原始观测字节。正常环境下这些字节必须全为零，

template <typename Provider>
concept ProviderLike = requires {
    { Provider::OBSERVATION_SIZE } -> std::convertible_to<std::size_t>;
    { Provider::observe() } -> std::same_as<Observation>;
};
```

`include/veil/crypto/sha256.hpp`：

```cpp
#pragma once

#include "../config.hpp"
#include "../detail/common.hpp"

#include <cstddef>
#include <cstdint>

namespace veil {

class Sha256 {
public:
    static constexpr std::size_t DIGEST_SIZE = 32;
    static constexpr std::size_t BLOCK_SIZE = 64;

    void init() noexcept {
        h_[0] = 0x6a09e667u; h_[1] = 0xbb67ae85u;
        h_[2] = 0x3c6ef372u; h_[3] = 0xa54ff53au;
        h_[4] = 0x510e527fu; h_[5] = 0x9b05688cu;
        h_[6] = 0x1f83d9abu; h_[7] = 0x5be0cd19u;
        total_ = 0;
        fill_ = 0;
    }

    void update(const void* data, std::size_t size) noexcept {
        const auto* p = static_cast<const std::uint8_t*>(data);
        total_ += size;

        while (size > 0) {
            const std::size_t take =
                (size < BLOCK_SIZE - fill_) ? size : (BLOCK_SIZE - fill_);
            for (std::size_t i = 0; i < take; ++i)
                buffer_[fill_ + i] = p[i];
            fill_ += take;
            p += take;
            size -= take;

            if (fill_ == BLOCK_SIZE) {
                transform(buffer_);
                fill_ = 0;
            }
        }
    }

    void final(std::uint8_t out[DIGEST_SIZE]) noexcept {
        const std::uint64_t bit_length = total_ * 8;

        buffer_[fill_++] = 0x80;
        if (fill_ > 56) {
            while (fill_ < BLOCK_SIZE)
                buffer_[fill_++] = 0;
            transform(buffer_);
            fill_ = 0;
        }
        while (fill_ < 56)
            buffer_[fill_++] = 0;

        for (std::size_t i = 0; i < 8; ++i)
            buffer_[56 + i] = static_cast<std::uint8_t>(bit_length >> ((7 - i) * 8));
        transform(buffer_);

        for (std::size_t i = 0; i < 8; ++i) {
            out[i * 4] = static_cast<std::uint8_t>(h_[i] >> 24);
            out[i * 4 + 1] = static_cast<std::uint8_t>(h_[i] >> 16);
            out[i * 4 + 2] = static_cast<std::uint8_t>(h_[i] >> 8);
            out[i * 4 + 3] = static_cast<std::uint8_t>(h_[i]);
        }
    }

    static void hash(const void* data, std::size_t size,
                     std::uint8_t out[DIGEST_SIZE]) noexcept {
        Sha256 ctx;
        ctx.init();
        ctx.update(data, size);
        ctx.final(out);
    }

private:
    VEIL_FORCEINLINE static std::uint32_t rotr(std::uint32_t v, int c) noexcept {
        return (v >> c) | (v << (32 - c));
    }

    void transform(const std::uint8_t block[BLOCK_SIZE]) noexcept {
        static const std::uint32_t k[64] = {
```

```cpp
#pragma once

#include "sha256.hpp"

#include <cstddef>
#include <cstdint>

namespace veil {

class Hmac {
public:
    static void compute(const void* key, std::size_t key_len,
                        const void* data, std::size_t data_len,
                        std::uint8_t out[Sha256::DIGEST_SIZE]) noexcept {
        std::uint8_t normalized[Sha256::BLOCK_SIZE]{};
        if (key_len > Sha256::BLOCK_SIZE) {
            Sha256::hash(key, key_len, normalized);
        } else if (key_len > 0) {
            const auto* p = static_cast<const std::uint8_t*>(key);
            for (std::size_t i = 0; i < key_len; ++i)
                normalized[i] = p[i];
        }

        std::uint8_t ipad[Sha256::BLOCK_SIZE];
        std::uint8_t opad[Sha256::BLOCK_SIZE];
        for (std::size_t i = 0; i < Sha256::BLOCK_SIZE; ++i) {
            ipad[i] = static_cast<std::uint8_t>(normalized[i] ^ 0x36);
            opad[i] = static_cast<std::uint8_t>(normalized[i] ^ 0x5c);
        }

        std::uint8_t inner[Sha256::DIGEST_SIZE];
        Sha256 ctx;
        ctx.init();
        ctx.update(ipad, Sha256::BLOCK_SIZE);
        ctx.update(data, data_len);
        ctx.final(inner);

        ctx.init();
        ctx.update(opad, Sha256::BLOCK_SIZE);
        ctx.update(inner, Sha256::DIGEST_SIZE);
        ctx.final(out);

        detail::secure_wipe(normalized, sizeof(normalized));
        detail::secure_wipe(ipad, sizeof(ipad));
        detail::secure_wipe(opad, sizeof(opad));
        detail::secure_wipe(inner, sizeof(inner));
    }
};
```

```cpp
#pragma once

#include "hmac.hpp"
#include "sha256.hpp"

#include <cstddef>
#include <cstdint>

namespace veil {

// RFC 5869。expand 阶段必须以 PRK 作为 HMAC 密钥，退回裸 SHA256
// 会让不同盐派生出相同的密钥流。
inline void hkdf_sha256(const void* ikm, std::size_t ikm_len,
                        const void* salt, std::size_t salt_len,
                        const void* info, std::size_t info_len,
                        void* out, std::size_t out_len) noexcept {
    std::uint8_t prk[Sha256::DIGEST_SIZE];
    Hmac::compute(salt_len > 0 ? salt : ikm, salt_len > 0 ? salt_len : ikm_len,
                  ikm, ikm_len, prk);

    auto* output = static_cast<std::uint8_t*>(out);
    std::uint8_t block[Sha256::DIGEST_SIZE];
    std::uint8_t previous[Sha256::DIGEST_SIZE];
    std::size_t previous_len = 0;
    std::uint8_t counter = 1;

    while (out_len > 0) {
        // info 上限 1024 覆盖 veil 的最大证据 blob，超出即截断并在文档中声明
        std::uint8_t message[Sha256::DIGEST_SIZE + 1024 + 1];
        std::size_t message_len = 0;
        for (std::size_t i = 0; i < previous_len; ++i)
            message[message_len++] = previous[i];

        const std::size_t info_copy = info_len < 1024 ? info_len : 1024;
        const auto* info_bytes = static_cast<const std::uint8_t*>(info);
        for (std::size_t i = 0; i < info_copy; ++i)
            message[message_len++] = info_bytes[i];

        message[message_len++] = counter;

        Hmac::compute(prk, Sha256::DIGEST_SIZE, message, message_len, block);

        const std::size_t take =
            out_len < Sha256::DIGEST_SIZE ? out_len : Sha256::DIGEST_SIZE;
        for (std::size_t i = 0; i < take; ++i)
            output[i] = block[i];

        for (std::size_t i = 0; i < Sha256::DIGEST_SIZE; ++i)
            previous[i] = block[i];
        previous_len = Sha256::DIGEST_SIZE;

        output += take;
        out_len -= take;
        ++counter;
    }

    detail::secure_wipe(prk, sizeof(prk));
    detail::secure_wipe(block, sizeof(block));
    detail::secure_wipe(previous, sizeof(previous));
}

}  // namespace veil
```

```cpp
#pragma once

#include "../config.hpp"
#include "../detail/common.hpp"

#include <cstddef>
#include <cstdint>

namespace veil {

// AES-256 块加密。S 盒由 GF(2^8) 逆元现算，二进制中不留 256 字节常量表。
class Aes256 {
public:
    static constexpr std::size_t BLOCK_SIZE = 16;
    static constexpr std::size_t KEY_SIZE = 32;
    static constexpr std::size_t ROUNDS = 14;

    void set_key(const std::uint8_t key[KEY_SIZE]) noexcept {
        expand_key(key);
        ready_ = true;
    }

    void encrypt_block(const std::uint8_t in[BLOCK_SIZE],
                       std::uint8_t out[BLOCK_SIZE]) const noexcept {
        std::uint8_t state[BLOCK_SIZE];
        for (std::size_t i = 0; i < BLOCK_SIZE; ++i)
            state[i] = static_cast<std::uint8_t>(in[i] ^ round_keys_[i]);

        for (std::size_t round = 1; round < ROUNDS; ++round) {
            sub_bytes(state);
            shift_rows(state);
            mix_columns(state);
            add_round_key(state, round);
        }

        sub_bytes(state);
        shift_rows(state);
        add_round_key(state, ROUNDS);

        for (std::size_t i = 0; i < BLOCK_SIZE; ++i)
            out[i] = state[i];

        detail::secure_wipe(state, sizeof(state));
    }

private:
    static constexpr std::uint8_t gmul(std::uint8_t a, std::uint8_t b) noexcept {
        std::uint8_t result = 0;
        for (int i = 0; i < 8; ++i) {
            if (b & 1)
                result ^= a;
            const std::uint8_t high = static_cast<std::uint8_t>(a & 0x80);
            a = static_cast<std::uint8_t>(a << 1);
            if (high)
                a ^= 0x1B;
            b = static_cast<std::uint8_t>(b >> 1);
        }
        return result;
    }

    static constexpr std::uint8_t inverse(std::uint8_t x) noexcept {
        if (x == 0)
            return 0;
        // 费马小定理：a^254 是 GF(2^8) 下的乘法逆元
        std::uint8_t result = 1;
        std::uint8_t base = x;
        unsigned int exp = 254;
        while (exp > 0) {
            if (exp & 1)
                result = gmul(result, base);
            base = gmul(base, base);
            exp >>= 1;
        }
        return result;
    }

    static constexpr std::uint8_t sbox(std::uint8_t x) noexcept {
        std::uint8_t inv = inverse(x);
        std::uint8_t y = inv;
        for (int i = 0; i < 4; ++i) {
            inv = static_cast<std::uint8_t>(((inv << 1) | (inv >> 7)) & 0xFF);
            y ^= inv;
        }
        return static_cast<std::uint8_t>(y ^ 0x63);
    }

    void expand_key(const std::uint8_t key[KEY_SIZE]) noexcept {
        constexpr std::size_t words = 4 * (ROUNDS + 1);

        std::uint8_t w[words * 4];
        for (std::size_t i = 0; i < KEY_SIZE; ++i)
            w[i] = key[i];

        std::uint8_t rcon = 0x01;
        for (std::size_t i = KEY_SIZE; i < words * 4; i += 4) {
            std::uint8_t temp[4] = {w[i - 4], w[i - 3], w[i - 2], w[i - 1]};

            if ((i / 4) % 8 == 0) {
                const std::uint8_t first = temp[0];
                temp[0] = static_cast<std::uint8_t>(sbox(temp[1]) ^ rcon);
                temp[1] = sbox(temp[2]);
                temp[2] = sbox(temp[3]);
                temp[3] = sbox(first);
                rcon = static_cast<std::uint8_t>(((rcon << 1) | (rcon >> 7)) & 0xFF);
                if (rcon == 0x1B)
                    rcon = 0x01;
            } else if ((i / 4) % 8 == 4) {
                for (std::size_t k = 0; k < 4; ++k)
                    temp[k] = sbox(temp[k]);
            }

            for (std::size_t k = 0; k < 4; ++k)
                w[i + k] = static_cast<std::uint8_t>(w[i - KEY_SIZE + k] ^ temp[k]);
        }

        for (std::size_t i = 0; i < BLOCK_SIZE; ++i)
            for (std::size_t k = 0; k < ROUNDS + 1; ++k)
                round_keys_[k * BLOCK_SIZE + i] = w[k * 16 + i];

        detail::secure_wipe(w, sizeof(w));
    }

    static void sub_bytes(std::uint8_t state[BLOCK_SIZE]) noexcept {
        for (std::size_t i = 0; i < BLOCK_SIZE; ++i)
            state[i] = sbox(state[i]);
    }

    static void shift_rows(std::uint8_t state[BLOCK_SIZE]) noexcept {
        std::uint8_t temp[BLOCK_SIZE];
        for (std::size_t c = 0; c < 4; ++c)
            for (std::size_t r = 0; r < 4; ++r)
                temp[c * 4 + r] = state[((c + r) % 4) * 4 + r];
        for (std::size_t i = 0; i < BLOCK_SIZE; ++i)
            state[i] = temp[i];
    }

    static void mix_columns(std::uint8_t state[BLOCK_SIZE]) noexcept {
        for (std::size_t c = 0; c < 4; ++c) {
            std::uint8_t* col = &state[c * 4];
            const std::uint8_t a0 = col[0], a1 = col[1], a2 = col[2], a3 = col[3];
            col[0] = static_cast<std::uint8_t>(gmul(a0, 2) ^ gmul(a1, 3) ^ a2 ^ a3);
            col[1] = static_cast<std::uint8_t>(a0 ^ gmul(a1, 2) ^ gmul(a2, 3) ^ a3);
            col[2] = static_cast<std::uint8_t>(a0 ^ a1 ^ gmul(a2, 2) ^ gmul(a3, 3));
            col[3] = static_cast<std::uint8_t>(gmul(a0, 3) ^ a1 ^ a2 ^ gmul(a3, 2));
        }
    }

    void add_round_key(std::uint8_t state[BLOCK_SIZE], std::size_t round) const noexcept {
        for (std::size_t i = 0; i < BLOCK_SIZE; ++i)
            state[i] ^= round_keys_[round * BLOCK_SIZE + i];
    }

    std::uint8_t round_keys_[(ROUNDS + 1) * BLOCK_SIZE]{};
    bool ready_ = false;
};

}  // namespace veil
```

```cpp
#pragma once

#include "aes.hpp"

#include <cstddef>
#include <cstdint>

namespace veil {

// AES-256-GCM。标签校验是解密的前置条件：不匹配时不写入任何输出，
// 不存在「先比对再解密」留下的可跳过窗口。
class Aes256Gcm {
public:
    static constexpr std::size_t KEY_SIZE = 32;
    static constexpr std::size_t NONCE_SIZE = 12;
    static constexpr std::size_t TAG_SIZE = 16;

    void set_key(const std::uint8_t key[KEY_SIZE]) noexcept {
        cipher_.set_key(key);
        derive_h();
        ready_ = true;
    }

    bool decrypt(const std::uint8_t* nonce,
                 const std::uint8_t* ciphertext, std::size_t length,
                 const std::uint8_t tag[TAG_SIZE],
                 const std::uint8_t* aad, std::size_t aad_len,
                 std::uint8_t* out) noexcept {
        if (!ready_)
            return false;

        std::uint8_t expected[TAG_SIZE];
        compute_tag(nonce, ciphertext, length, aad, aad_len, expected);

        if (detail::constant_time_compare(expected, tag, TAG_SIZE) == 0) {
            detail::secure_wipe(expected, sizeof(expected));
            return false;
        }
        detail::secure_wipe(expected, sizeof(expected));

        ctr_crypt(nonce, ciphertext, length, out);
        return true;
    }

    bool encrypt(const std::uint8_t* nonce,
                 const std::uint8_t* plaintext, std::size_t length,
                 const std::uint8_t* aad, std::size_t aad_len,
                 std::uint8_t* out, std::uint8_t tag[TAG_SIZE]) noexcept {
        if (!ready_)
            return false;

        ctr_crypt(nonce, plaintext, length, out);
        compute_tag(nonce, out, length, aad, aad_len, tag);
        return true;
    }

private:
    void derive_h() noexcept {
        std::uint8_t zero[16]{};
        std::uint8_t h[16];
        cipher_.encrypt_block(zero, h);
        for (std::size_t i = 0; i < 16; ++i)
            h_[i] = h[i];
        detail::secure_wipe(h, sizeof(h));
    }

    void ctr_crypt(const std::uint8_t* nonce,
                   const std::uint8_t* input, std::size_t length,
                   std::uint8_t* out) noexcept {
        std::uint8_t counter[16]{};
        for (std::size_t i = 0; i < NONCE_SIZE; ++i)
            counter[i] = nonce[i];
        // 96 位 nonce 下 J0 = nonce || 0x00000001，数据加密自 inc32(J0) 起算。
        // 起始值取 1 会与标签掩码的 E_K(J0) 重合，导致密文整体偏移一块。
        counter[15] = 2;

        std::uint8_t keystream[16];
        std::size_t offset = 0;

        while (offset < length) {
            cipher_.encrypt_block(counter, keystream);

            const std::size_t take = (length - offset < 16) ? (length - offset) : 16;
            for (std::size_t i = 0; i < take; ++i)
                out[offset + i] = static_cast<std::uint8_t>(input[offset + i] ^ keystream[i]);

            for (int i = 15; i >= 12; --i) {
                if (++counter[i] != 0)
                    break;
            }

            offset += take;
        }

        detail::secure_wipe(counter, sizeof(counter));
        detail::secure_wipe(keystream, sizeof(keystream));
    }

    void compute_tag(const std::uint8_t* nonce,
                     const std::uint8_t* data, std::size_t length,
                     const std::uint8_t* aad, std::size_t aad_len,
                     std::uint8_t tag[TAG_SIZE]) const noexcept {
        std::uint8_t state[16]{};

        if (aad != nullptr && aad_len > 0)
            ghash(state, aad, aad_len);

        if (data != nullptr && length > 0)
            ghash(state, data, length);

        std::uint8_t lengths[16]{};
        const std::uint64_t aad_bits = static_cast<std::uint64_t>(aad_len) * 8;
        const std::uint64_t data_bits = static_cast<std::uint64_t>(length) * 8;
        for (std::size_t i = 0; i < 8; ++i) {
            lengths[i] = static_cast<std::uint8_t>(aad_bits >> ((7 - i) * 8));
            lengths[8 + i] = static_cast<std::uint8_t>(data_bits >> ((7 - i) * 8));
        }
        xor_block(state, lengths);
        multiply_h(state);

        std::uint8_t mask[16];
        std::uint8_t counter[16]{};
        for (std::size_t i = 0; i < NONCE_SIZE; ++i)
            counter[i] = nonce[i];
        counter[15] = 1;
        cipher_.encrypt_block(counter, mask);

        for (std::size_t i = 0; i < TAG_SIZE; ++i)
            tag[i] = static_cast<std::uint8_t>(state[i] ^ mask[i]);

        detail::secure_wipe(state, sizeof(state));
        detail::secure_wipe(lengths, sizeof(lengths));
        detail::secure_wipe(mask, sizeof(mask));
    }

    void ghash(std::uint8_t state[16], const std::uint8_t* data,
               std::size_t length) const noexcept {
        std::size_t offset = 0;
        while (offset < length) {
            const std::size_t take = (length - offset < 16) ? (length - offset) : 16;
            std::uint8_t block[16]{};
            for (std::size_t i = 0; i < take; ++i)
                block[i] = data[offset + i];

            xor_block(state, block);
            multiply_h(state);

            offset += take;
        }
    }

    static void xor_block(std::uint8_t state[16], const std::uint8_t block[16]) noexcept {
        for (std::size_t i = 0; i < 16; ++i)
            state[i] ^= block[i];
    }

    void multiply_h(std::uint8_t state[16]) const noexcept {
        std::uint8_t z[16]{};
        std::uint8_t v[16];
        for (std::size_t i = 0; i < 16; ++i)
            v[i] = h_[i];

        for (std::size_t i = 0; i < 128; ++i) {
            const std::size_t byte_index = i / 8;
            const std::size_t bit_index = 7 - (i % 8);
            if ((state[byte_index] >> bit_index) & 1) {
                for (std::size_t k = 0; k < 16; ++k)
                    z[k] ^= v[k];
            }

            const bool lsb = (v[15] & 1) != 0;
            for (int k = 15; k > 0; --k)
                v[k] = static_cast<std::uint8_t>((v[k] >> 1) | ((v[k - 1] & 1) << 7));
            v[0] = static_cast<std::uint8_t>(v[0] >> 1);
            if (lsb)
                v[0] ^= 0xE1;
        }

        for (std::size_t i = 0; i < 16; ++i)
            state[i] = z[i];
    }

    Aes256 cipher_;
    std::uint8_t h_[16]{};
    bool ready_ = false;
};

}  // namespace veil
```

```cpp
#include "../config.hpp"
#include "../detail/common.hpp"

#include <cstddef>
#include <cstdint>

extern "C" {
// Monocypher 是 C99 代码，需以 C 链接方式引入，避免 C++ 名称修饰
// 导致链接期找不到符号。
int crypto_ed25519_check(const uint8_t signature[64],
                         const uint8_t public_key[32],
                         const uint8_t* message, size_t message_size);
}

namespace veil {

// RFC 8032 验签。系统 CNG 不提供 Ed25519，故由 Monocypher 承担。
// 选 RFC 8032 而非 Monocypher 顶层 BLAKE2b EdDSA 变体，是因为服务端
// Node crypto.sign(null, ...) 产出 SHA-512 版本，两者不互通。
class Ed25519 {
public:
    static constexpr std::size_t PUBLIC_KEY_SIZE = 32;
    static constexpr std::size_t SIGNATURE_SIZE = 64;

    bool set_public_key(const std::uint8_t* public_key,
                        std::size_t length) noexcept {
        if (public_key == nullptr || length != PUBLIC_KEY_SIZE) {
            loaded_ = false;
            return false;
        }

        for (std::size_t i = 0; i < PUBLIC_KEY_SIZE; ++i)
            public_key_[i] = public_key[i];
        loaded_ = true;
        return true;
    }

    bool set_public_key_hex(const char* hex) noexcept {
        if (hex == nullptr)
            return false;

        std::uint8_t decoded[PUBLIC_KEY_SIZE];
        for (std::size_t i = 0; i < PUBLIC_KEY_SIZE; ++i) {
            const int high = hex_nibble(hex[i * 2]);
            const int low = hex_nibble(hex[i * 2 + 1]);
            if (high < 0 || low < 0) {
                detail::secure_wipe(decoded, sizeof(decoded));
                loaded_ = false;
                return false;
            }
            decoded[i] = static_cast<std::uint8_t>((high << 4) | low);
        }

        const bool ok = set_public_key(decoded, sizeof(decoded));
        detail::secure_wipe(decoded, sizeof(decoded));
        return ok;
    }

    bool loaded() const noexcept { return loaded_; }

    bool verify(const std::uint8_t* message, std::size_t message_length,
                const std::uint8_t* signature,
                std::size_t signature_length) const noexcept {
        if (!loaded_ || signature == nullptr ||
            signature_length != SIGNATURE_SIZE)
            return false;

        if (message == nullptr && message_length != 0)
            return false;

        return crypto_ed25519_check(signature, public_key_,
                                    message, message_length) == 0;
    }

    bool verify_hex(const std::uint8_t* message, std::size_t message_length,
                    const char* signature_hex) const noexcept {
        if (signature_hex == nullptr)
            return false;

        std::uint8_t decoded[SIGNATURE_SIZE];
        for (std::size_t i = 0; i < SIGNATURE_SIZE; ++i) {
            const int high = hex_nibble(signature_hex[i * 2]);
            const int low = hex_nibble(signature_hex[i * 2 + 1]);
            if (high < 0 || low < 0) {
                detail::secure_wipe(decoded, sizeof(decoded));
                return false;
            }
            decoded[i] = static_cast<std::uint8_t>((high << 4) | low);
        }

        const bool ok = verify(message, message_length, decoded, sizeof(decoded));
        detail::secure_wipe(decoded, sizeof(decoded));
        return ok;
    }

    ~Ed25519() noexcept { detail::secure_wipe(public_key_, sizeof(public_key_)); }

    Ed25519() = default;
    Ed25519(const Ed25519&) = delete;
    Ed25519& operator=(const Ed25519&) = delete;

private:
    static int hex_nibble(char c) noexcept {
        if (c >= '0' && c <= '9') return c - '0';
        if (c >= 'a' && c <= 'f') return c - 'a' + 10;
        if (c >= 'A' && c <= 'F') return c - 'A' + 10;
        return -1;
    }

    std::uint8_t public_key_[PUBLIC_KEY_SIZE]{};
    bool loaded_ = false;
};

}  // namespace veil
```

```cpp
#pragma once

#include "../config.hpp"
#include "../detail/mix.hpp"

#include <cstddef>
#include <cstdint>

#include <windows.h>

namespace veil::platform {

// 从 ntdll 导出表按名字取 SSN。
//
// 直接读函数体前 4 字节会在 stub 被 inline hook 时读到跳转而不是
// `mov r10, rcx; mov eax, SSN`。这里不读 stub，改为按导出表名字顺序
// 推导：ntdll 的导出名表按字母序排列，SSN 等于该名字在导出名表中的
// 序号，此性质在 Win10 1809 之后稳定成立。
//
// 该性质不适用于被人工重排的 ntdll，故对推导结果做范围校验。
inline std::uint32_t syscall_number(const char* export_name) noexcept {
    if (export_name == nullptr)
        return 0;

    HMODULE ntdll = GetModuleHandleA("ntdll.dll");
    if (ntdll == nullptr)
        return 0;

    const auto* base = reinterpret_cast<const std::uint8_t*>(ntdll);
    const auto* dos = reinterpret_cast<const IMAGE_DOS_HEADER*>(base);
    if (dos->e_magic != IMAGE_DOS_SIGNATURE)
        return 0;

    const auto* nt = reinterpret_cast<const IMAGE_NT_HEADERS*>(
        base + static_cast<std::ptrdiff_t>(dos->e_lfanew));
    if (nt->Signature != IMAGE_NT_SIGNATURE)
        return 0;

    const IMAGE_DATA_DIRECTORY dir =
        nt->OptionalHeader.DataDirectory[IMAGE_DIRECTORY_ENTRY_EXPORT];
    if (dir.VirtualAddress == 0 || dir.Size == 0)
        return 0;

    const auto* exports = reinterpret_cast<const IMAGE_EXPORT_DIRECTORY*>(
        base + dir.VirtualAddress);
    const auto* names = reinterpret_cast<const std::uint32_t*>(
        base + exports->AddressOfNames);

    for (std::uint32_t i = 0; i < exports->NumberOfNames; ++i) {
        const char* name = reinterpret_cast<const char*>(base + names[i]);
        std::size_t k = 0;
        while (export_name[k] != '\0' && name[k] == export_name[k])
            ++k;
        if (export_name[k] == '\0' && name[k] == '\0')
            return i;
    }
    return 0;
}

// x64 syscall 直发。参数按 Windows x64 调用约定传入：
// rcx, rdx, r8, r9，其余压栈。
//
// 前四个参数足以覆盖 veil 使用的全部系统调用；需要更多参数的场景
// 由具体包装函数自行处理。
inline long syscall_direct(std::uint32_t number,
                           std::uintptr_t a1, std::uintptr_t a2,
                           std::uintptr_t a3, std::uintptr_t a4,
                           std::uintptr_t a5) noexcept {
#if defined(_MSC_VER)
    // MSVC 不支持 x64 内联汇编，改用裸函数声明由链接器提供。
    // 实现位于 platform/syscall_stub.asm。
    extern long veil_syscall_stub(std::uint32_t, std::uintptr_t,
                                  std::uintptr_t, std::uintptr_t,
                                  std::uintptr_t, std::uintptr_t);
    return veil_syscall_stub(number, a1, a2, a3, a4, a5);
#else
    register std::uintptr_t r10 asm("r10") = a4;
    register std::uintptr_t r8 asm("r8") = a3;
    register std::uintptr_t r9 asm("r9") = a5;
    register std::uintptr_t rdx asm("rdx") = a2;
    register std::uintptr_t rcx asm("rcx") = a1;
```

```cpp
#pragma once

#include "../config.hpp"

#include <cstddef>
#include <cstdint>

#include <winternl.h>

namespace veil::platform {

// 部分 SDK 未导出 LDR_DATA_TABLE_ENTRY 的完整字段布局，
// 这里按公开的 PEB 结构自行声明，避免依赖工具链差异。
struct LdrEntry {
    LIST_ENTRY InLoadOrderLinks;
    LIST_ENTRY InMemoryOrderLinks;
    LIST_ENTRY InInitializationOrderLinks;
    void* DllBase;
    void* EntryPoint;
    ULONG SizeOfImage;
    UNICODE_STRING FullDllName;
    UNICODE_STRING BaseDllName;
};

struct ModuleInfo {
    void* base;
    std::uint32_t size;
    const wchar_t* path;
    std::uint32_t path_length;
};

VEIL_FORCEINLINE PPEB peb() noexcept {
    return reinterpret_cast<PPEB>(__readgsqword(0x60));
}

inline std::size_t module_count() noexcept {
    PPEB p = peb();
    if (p == nullptr || p->Ldr == nullptr)
        return 0;

    std::size_t count = 0;
    LIST_ENTRY* head = &p->Ldr->InMemoryOrderModuleList;
    for (LIST_ENTRY* entry = head->Flink; entry != head && entry != nullptr;
         entry = entry->Flink) {
        if (++count > 4096)
            break;
    }
    return count;
}

inline bool module_at(std::size_t index, ModuleInfo* out) noexcept {
    if (out == nullptr)
        return false;

    PPEB p = peb();
    if (p == nullptr || p->Ldr == nullptr)
        return false;

    std::size_t cursor = 0;
    LIST_ENTRY* head = &p->Ldr->InMemoryOrderModuleList;
    for (LIST_ENTRY* entry = head->Flink; entry != head && entry != nullptr;
         entry = entry->Flink) {
        if (cursor == index) {
            auto* node = CONTAINING_RECORD(entry, LdrEntry, InMemoryOrderLinks);
            out->base = node->DllBase;
            out->size = node->SizeOfImage;
            out->path = node->FullDllName.Buffer;
            out->path_length = node->FullDllName.Length / sizeof(wchar_t);
            return out->base != nullptr && out->path != nullptr;
        }
        ++cursor;
        if (cursor > 4096)
            break;
    }
    return false;
}

inline wchar_t lower_ascii(wchar_t c) noexcept {
    return (c >= L'A' && c <= L'Z') ? static_cast<wchar_t>(c + (L'a' - L'A')) : c;
}

inline bool contains_ci(const wchar_t* haystack, std::uint32_t haystack_len,
                        const wchar_t* needle) noexcept {
    if (haystack == nullptr || needle == nullptr)
        return false;

    std::uint32_t needle_len = 0;
    while (needle[needle_len] != L'\0')
        ++needle_len;
    if (needle_len == 0 || needle_len > haystack_len)
        return false;

    for (std::uint32_t start = 0; start + needle_len <= haystack_len; ++start) {
        bool matched = true;
        for (std::uint32_t i = 0; i < needle_len; ++i) {
            if (lower_ascii(haystack[start + i]) != lower_ascii(needle[i])) {
                matched = false;
                break;
            }
        }
        if (matched)
            return true;
    }
    return false;
}

inline bool starts_with_ci(const wchar_t* path, std::uint32_t path_len,
                           const wchar_t* prefix) noexcept {
    if (path == nullptr || prefix == nullptr)
        return false;

    std::uint32_t prefix_len = 0;
    while (prefix[prefix_len] != L'\0')
        ++prefix_len;
    if (prefix_len > path_len)
        return false;

    for (std::uint32_t i = 0; i < prefix_len; ++i) {
        if (lower_ascii(path[i]) != lower_ascii(prefix[i]))
            return false;
    }
    return true;
}

inline void* module_base_ci(const wchar_t* name) noexcept {
    std::uint32_t name_len = 0;
    while (name[name_len] != L'\0')
        ++name_len;

    const std::size_t count = module_count();
    for (std::size_t i = 0; i < count; ++i) {
        ModuleInfo info{};
        if (!module_at(i, &info))
            continue;
        if (info.path_length < name_len)
            continue;

        // 只比较文件名部分，避免把路径中恰好包含目标名的目录误命中
        const wchar_t* tail = info.path + (info.path_length - name_len);
        bool matched = true;
        for (std::uint32_t k = 0; k < name_len; ++k) {
            if (lower_ascii(tail[k]) != lower_ascii(name[k])) {
                matched = false;
                break;
            }
        }
        if (matched)
            return info.base;
    }
    return nullptr;
}

}  // namespace veil::platform
```

```cpp
#pragma once

#include "../config.hpp"
#include "syscall.hpp"

#include <cstddef>
#include <cstdint>

#include <windows.h>

namespace veil::platform {

struct RegionInfo {
    void* base;
    std::uintptr_t allocation_base;
    std::uint32_t size;
    std::uint32_t state;
    std::uint32_t protect;
    std::uint32_t type;
};

// 走 NtQueryVirtualMemory 而非 VirtualQuery。后者是 kernel32 导出，
// 改写其 IAT 槽位即可让全部区间内省返回伪造结果。
inline bool query_region(const void* address, RegionInfo* out) noexcept {
    if (address == nullptr || out == nullptr)
        return false;

    static const std::uint32_t ssn = syscall_number("NtQueryVirtualMemory");
    if (ssn == 0)
        return false;

    struct MemoryBasicInformation {
        void* base;
        void* allocation_base;
        std::uint32_t allocation_protect;
        std::uint32_t partition_id;
        std::uintptr_t region_size;
        std::uint32_t state;
        std::uint32_t protect;
        std::uint32_t type;
    };

    MemoryBasicInformation mbi{};
    unsigned long returned = 0;
    const long status = syscall_direct(
        ssn,
        static_cast<std::uintptr_t>(reinterpret_cast<std::uintptr_t>(GetCurrentProcess())),
        reinterpret_cast<std::uintptr_t>(address),
        0,  // MemoryBasicInformation
        reinterpret_cast<std::uintptr_t>(&mbi),
        sizeof(mbi));

    if (status != 0 || returned < sizeof(mbi))
        return false;

    out->base = mbi.base;
    out->allocation_base = reinterpret_cast<std::uintptr_t>(mbi.allocation_base);
    out->size = static_cast<std::uint32_t>(mbi.region_size);
    out->state = mbi.state;
    out->protect = mbi.protect;
    out->type = mbi.type;
    return true;
}

// 页面是否具备写权限。供 provider 判定区间属性。
inline bool is_writable(const RegionInfo& info) noexcept {
    const std::uint32_t p = info.protect & 0xFFu;
    return p == PAGE_READWRITE || p == PAGE_WRITECOPY ||
           p == PAGE_EXECUTE_READWRITE || p == PAGE_EXECUTE_WRITECOPY;
}

}  // namespace veil::platform
```

```cpp
#pragma once

#include "../config.hpp"
#include "../crypto/sha256.hpp"
#include "../detail/common.hpp"

#include <cstddef>
#include <cstdint>

#include <windows.h>

namespace veil::platform {

struct SectionView {
    const char* name;
    std::uint32_t virtual_address;
    std::uint32_t virtual_size;
    std::uint32_t raw_offset;
    std::uint32_t raw_size;
    std::uint32_t characteristics;
    bool executable;
};

// 按磁盘读取自身 PE 映像，供完整性 provider 取 .text 原始字节。
//
// 内存映像可能已被改写，磁盘侧则只有修改文件才能改变，而修改文件
// 需要重新签名。两者摘要的差值因此是一个可信度更高的基线。
class ImageFile {
public:
    bool open_self() noexcept {
        const HMODULE module = GetModuleHandleA(nullptr);
        if (module == nullptr)
            return false;
        return open_module(module);
    }

    bool open_module(void* base) noexcept {
        close();
        if (base == nullptr)
            return false;

        image_base_ = base;

        char path[MAX_PATH];
        const DWORD length = GetModuleFileNameA(static_cast<HMODULE>(base),
                                                path, MAX_PATH);
        if (length == 0 || length >= MAX_PATH)
            return false;

        file_ = CreateFileA(path, GENERIC_READ, FILE_SHARE_READ | FILE_SHARE_WRITE,
                            nullptr, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, nullptr);
        if (file_ == INVALID_HANDLE_VALUE)
            return false;

        const auto* dos = static_cast<const IMAGE_DOS_HEADER*>(base);
        if (dos->e_magic != IMAGE_DOS_SIGNATURE)
            return false;

        const auto* nt = reinterpret_cast<const IMAGE_NT_HEADERS*>(
            static_cast<const std::uint8_t*>(base) + dos->e_lfanew);
        if (nt->Signature != IMAGE_NT_SIGNATURE)
            return false;

        const IMAGE_FILE_HEADER& header = nt->FileHeader;
        if (header.NumberOfSections == 0 ||
            header.NumberOfSections > MAX_SECTIONS)
            return false;

        section_count_ = header.NumberOfSections;
        for (std::size_t i = 0; i < section_count_; ++i) {
            const IMAGE_SECTION_HEADER& raw = nt->SectionHeader[i];
            sections_[i].name = reinterpret_cast<const char*>(raw.Name);
            sections_[i].virtual_address = raw.VirtualAddress;
            sections_[i].virtual_size = raw.Misc.VirtualSize;
            sections_[i].raw_offset = raw.PointerToRawData;
            sections_[i].raw_size = raw.SizeOfRawData;
            sections_[i].characteristics = raw.Characteristics;
            sections_[i].executable =
                (raw.Characteristics & IMAGE_SCN_MEM_EXECUTE) != 0;
        }

        return true;
    }

    void close() noexcept {
        if (file_ != INVALID_HANDLE_VALUE) {
            CloseHandle(file_);
            file_ = INVALID_HANDLE_VALUE;
        }
        image_base_ = nullptr;
        section_count_ = 0;
    }

    ~ImageFile() noexcept { close(); }

    ImageFile() = default;
    ImageFile(const ImageFile&) = delete;
    ImageFile& operator=(const ImageFile&) = delete;

    bool loaded() const noexcept { return file_ != INVALID_HANDLE_VALUE; }
    void* image_base() const noexcept { return image_base_; }
    std::size_t section_count() const noexcept { return section_count_; }

    bool section_at(std::size_t index, SectionView* out) const noexcept {
        if (out == nullptr || index >= section_count_)
            return false;
        *out = sections_[index];
        return true;
    }

    // 从磁盘读取该段的原始字节并求摘要。段在文件中按 FileAlignment 对齐，
    // 只取 SizeOfRawData 覆盖的部分。
    bool digest_disk(std::size_t index, std::uint8_t out[32]) noexcept {
        if (!loaded() || index >= section_count_ || out == nullptr)
            return false;

        const SectionView& section = sections_[index];
        if (section.raw_size == 0 || section.raw_size > MAX_SECTION_BYTES)
            return false;

        LARGE_INTEGER offset{};
        offset.QuadPart = section.raw_offset;
        if (!SetFilePointerEx(file_, offset, nullptr, FILE_BEGIN))
            return false;

        std::uint8_t buffer[CHUNK];
        Sha256 ctx;
        ctx.init();

        std::size_t remaining = section.raw_size;
        while (remaining > 0) {
            const DWORD want = static_cast<DWORD>(
                remaining < CHUNK ? remaining : CHUNK);
            DWORD read = 0;
            if (!ReadFile(file_, buffer, want, &read, nullptr) || read == 0)
                return false;
            ctx.update(buffer, read);
            remaining -= read;
        }

        ctx.final(out);
        detail::secure_wipe(buffer, sizeof(buffer));
        return true;
    }

    // 对内存映像中同一段求摘要。只取 VirtualSize 覆盖的部分，
    // 与磁盘侧对齐到 SizeOfRawData 的差异由调用方通过 region 登记消除。
    bool digest_memory(std::size_t index, std::uint8_t out[32]) const noexcept {
        if (image_base_ == nullptr || index >= section_count_ || out == nullptr)
            return false;

        const SectionView& section = sections_[index];
        const std::size_t size =
            section.raw_size < section.virtual_size ? section.raw_size
                                                    : section.virtual_size;
        if (size == 0)
            return false;

        const auto* bytes = static_cast<const std::uint8_t*>(image_base_) +
                            section.virtual_address;
        Sha256::hash(bytes, size, out);
        return true;
    }

private:
    static constexpr std::size_t MAX_SECTIONS = 96;
    static constexpr std::size_t MAX_SECTION_BYTES = 64u * 1024u * 1024u;
    static constexpr std::size_t CHUNK = 16u * 1024u;

    HANDLE file_ = INVALID_HANDLE_VALUE;
    void* image_base_ = nullptr;
    std::size_t section_count_ = 0;
    SectionView sections_[MAX_SECTIONS]{};
};

}  // namespace veil::platform
```

```cpp
#pragma once

#include "../config.hpp"
#include "../detail/common.hpp"
#include "../detail/mix.hpp"

#include <cstddef>
#include <cstdint>

#include <intrin.h>

namespace veil::providers {

// 四路物理熵混合。系统 CNG 被改写时仍能产出不可预测的密钥材料。
class Entropy {
public:
    static constexpr std::size_t KEY_SIZE = 32;
    static constexpr std::size_t NONCE_SIZE = 12;

    Entropy() noexcept { reseed(); }

    void reseed() noexcept {
        std::uint64_t sources[4] = {
            rdtsc(),
            rdrand(),
            query_counter(),
            address_jitter(),
        };

        std::uint8_t seed[KEY_SIZE];
        for (std::size_t i = 0; i < 4; ++i) {
            const std::uint64_t mixed =
                detail::mix64(sources[i] ^ (0x9E3779B97F4A7C15ULL * (i + 1)));
            for (std::size_t b = 0; b < 8; ++b)
                seed[i * 8 + b] = static_cast<std::uint8_t>(mixed >> (b * 8));
        }

        detail::secure_wipe(key_, KEY_SIZE);
        detail::secure_wipe(nonce_, NONCE_SIZE);

        for (std::size_t i = 0; i < KEY_SIZE; ++i)
            key_[i] = seed[i];
        for (std::size_t i = 0; i < NONCE_SIZE; ++i)
            nonce_[i] = seed[(i + 17) % KEY_SIZE] ^ static_cast<std::uint8_t>(i);
        for (std::size_t i = 0; i < 4; ++i)
            counter_[i] = static_cast<std::uint8_t>(seed[i] ^ seed[i + 23]);

        detail::secure_wipe(seed, KEY_SIZE);
        detail::secure_wipe(sources, sizeof(sources));
    }

    void generate(void* output, std::size_t size) noexcept {
        auto* out = static_cast<std::uint8_t*>(output);
        while (size > 0) {
            std::uint8_t block[64];
            chacha20_block(block);
            increment_counter();

            const std::size_t take = size < 64 ? size : 64;
            for (std::size_t i = 0; i < take; ++i)
                out[i] = block[i];
            detail::secure_wipe(block, 64);

            out += take;
            size -= take;
        }
    }

    ~Entropy() noexcept {
        detail::secure_wipe(key_, KEY_SIZE);
        detail::secure_wipe(nonce_, NONCE_SIZE);
        detail::secure_wipe(counter_, 4);
    }

    Entropy(const Entropy&) = delete;
    Entropy& operator=(const Entropy&) = delete;

private:
    static std::uint64_t rdtsc() noexcept {
        return __rdtsc();
    }

    // RDRAND 在部分虚拟化环境下被禁用。十六次重试后仍失败时回落到
    // 时间戳与计数器混合，而不是返回零值。
    static std::uint64_t rdrand() noexcept {
        std::uint64_t value = 0;
        unsigned char ok = 0;
        for (int attempt = 0; attempt < 16 && ok == 0; ++attempt)
            ok = static_cast<unsigned char>(_rdrand64_step(&value));

        if (ok == 0)
            return detail::mix64(__rdtsc() ^ 0xA5A5A5A5A5A5A5A5ULL);
        return value;
    }

    static std::uint64_t query_counter() noexcept {
        LARGE_INTEGER li{};
        QueryPerformanceCounter(&li);
        return static_cast<std::uint64_t>(li.QuadPart);
    }

    static std::uint64_t address_jitter() noexcept {
        std::uint64_t local = 0;
        void* heap = HeapAlloc(GetProcessHeap(), 0, 32);
        if (heap != nullptr)
            HeapFree(GetProcessHeap(), 0, heap);
        return detail::mix64(reinterpret_cast<std::uintptr_t>(&local) ^
                             reinterpret_cast<std::uintptr_t>(heap));
    }

    static std::uint32_t rotl32(std::uint32_t v, int c) noexcept {
        return (v << c) | (v >> (32 - c));
    }

    static void quarter_round(std::uint32_t& a, std::uint32_t& b,
                              std::uint32_t& c, std::uint32_t& d) noexcept {
        a += b; d ^= a; d = rotl32(d, 16);
        c += d; b ^= c; b = rotl32(b, 12);
        a += b; d ^= a; d = rotl32(d, 8);
        c += d; b ^= c; b = rotl32(b, 7);
    }

    void chacha20_block(std::uint8_t out[64]) const noexcept {
        std::uint32_t state[16] = {
            0x61707865u, 0x3320646eu, 0x79622d32u, 0x6b206574u,
        };

        for (std::size_t i = 0; i < 8; ++i) {
            state[4 + i] = static_cast<std::uint32_t>(key_[i * 4]) |
                           (static_cast<std::uint32_t>(key_[i * 4 + 1]) << 8) |
                           (static_cast<std::uint32_t>(key_[i * 4 + 2]) << 16) |
                           (static_cast<std::uint32_t>(key_[i * 4 + 3]) << 24);
        }

        state[12] = static_cast<std::uint32_t>(counter_[0]) |
                    (static_cast<std::uint32_t>(counter_[1]) << 8) |
                    (static_cast<std::uint32_t>(counter_[2]) << 16) |
                    (static_cast<std::uint32_t>(counter_[3]) << 24);

        state[13] = static_cast<std::uint32_t>(nonce_[0]) |
                    (static_cast<std::uint32_t>(nonce_[1]) << 8) |
                    (static_cast<std::uint32_t>(nonce_[2]) << 16) |
                    (static_cast<std::uint32_t>(nonce_[3]) << 24);
        state[14] = static_cast<std::uint32_t>(nonce_[4]) |
                    (static_cast<std::uint32_t>(nonce_[5]) << 8) |
                    (static_cast<std::uint32_t>(nonce_[6]) << 16) |
                    (static_cast<std::uint32_t>(nonce_[7]) << 24);
        state[15] = static_cast<std::uint32_t>(nonce_[8]) |
                    (static_cast<std::uint32_t>(nonce_[9]) << 8) |
                    (static_cast<std::uint32_t>(nonce_[10]) << 16) |
                    (static_cast<std::uint32_t>(nonce_[11]) << 24);

        std::uint32_t working[16];
        for (std::size_t i = 0; i < 16; ++i)
            working[i] = state[i];

        for (int round = 0; round < 10; ++round) {
            quarter_round(working[0], working[4], working[8], working[12]);
            quarter_round(working[1], working[5], working[9], working[13]);
            quarter_round(working[2], working[6], working[10], working[14]);
            quarter_round(working[3], working[7], working[11], working[15]);
            quarter_round(working[0], working[5], working[10], working[15]);
            quarter_round(working[1], working[6], working[11], working[12]);
            quarter_round(working[2], working[7], working[8], working[13]);
            quarter_round(working[3], working[4], working[9], working[14]);
        }

        for (std::size_t i = 0; i < 16; ++i) {
            const std::uint32_t v = working[i] + state[i];
            out[i * 4] = static_cast<std::uint8_t>(v);
            out[i * 4 + 1] = static_cast<std::uint8_t>(v >> 8);
            out[i * 4 + 2] = static_cast<std::uint8_t>(v >> 16);
            out[i * 4 + 3] = static_cast<std::uint8_t>(v >> 24);
        }

        detail::secure_wipe(state, sizeof(state));
        detail::secure_wipe(working, sizeof(working));
    }

    void increment_counter() noexcept {
        for (std::size_t i = 0; i < 4; ++i) {
            if (++counter_[i] != 0)
                break;
        }
    }

    std::uint8_t key_[KEY_SIZE]{};
    std::uint8_t nonce_[NONCE_SIZE]{};
    std::uint8_t counter_[4]{};
};

}  // namespace veil::providers
```

```cpp
#pragma once

#include "../config.hpp"
#include "../detail/observe.hpp"
#include "../platform/peb.hpp"
#include "../platform/syscall.hpp"

#include <cstddef>
#include <cstdint>

#include <intrin.h>

namespace veil::providers {

// 反调试观测。
//
// 与常规反调试库的区别：这里不返回「是否被调试」。字段是原始计数，
// 正常环境全为零，直接作为 HKDF 的 info 参与派生。攻击者要让派生成立，
// 必须让全部字段同时回到零值，等价于根本不调试。
//
// 没有可比对的布尔量，也就没有可单点改写的跳转。
struct AntiDebugSnapshot {
    std::uint8_t being_debugged;        // PEB.BeingDebugged，正常 0
    std::uint8_t hardware_breakpoints;  // Dr0-Dr3 中非零数量，正常 0
    std::uint8_t reserved0;
    std::uint8_t reserved1;
    std::uint32_t nt_global_flag;       // PEB.NtGlobalFlag 调试位，正常 0
    std::uint32_t veh_foreign_handlers; // 非系统模块注册的 VEH，阶段二填充
    std::uint32_t page_guard_events;    // PAGE_GUARD 事件计数，阶段二填充

    static constexpr std::size_t OBSERVATION_SIZE = 16;

    static Observation observe() noexcept {
        static thread_local AntiDebugSnapshot snapshot;
        snapshot = capture();
        return Observation{&snapshot, OBSERVATION_SIZE};
    }

    static AntiDebugSnapshot capture() noexcept {
        AntiDebugSnapshot snapshot{};

        PPEB p = platform::peb();
        if (p != nullptr) {
            snapshot.being_debugged = p->BeingDebugged;
            snapshot.nt_global_flag =
                read_global_flag(p) & DEBUG_FLAGS_MASK;
        }

        snapshot.hardware_breakpoints = count_hardware_breakpoints();
        return snapshot;
    }

private:
    static constexpr std::uint32_t DEBUG_FLAGS_MASK = 0x70u;

    static std::uint32_t read_global_flag(const PPEB p) noexcept {
        // NtGlobalFlag 在 x64 下位于 PEB + 0xBC。该偏移在 Win7 之后稳定。
        const auto* flags = reinterpret_cast<const std::uint32_t*>(
            reinterpret_cast<std::uintptr_t>(p) + 0xBC);
        return *flags;
    }

    // 读取当前线程的调试寄存器。硬件断点不需要改任何代码，因此不会
    // 触发完整性校验，只能通过 Dr 寄存器发现。
    static std::uint8_t count_hardware_breakpoints() noexcept {
        CONTEXT context{};
        context.ContextFlags = CONTEXT_DEBUG_REGISTERS;
        if (RtlCaptureContext(&context) == 0)
            return 0;

        std::uint8_t count = 0;
        if (context.Dr0 != 0) ++count;
        if (context.Dr1 != 0) ++count;
        if (context.Dr2 != 0) ++count;
        if (context.Dr3 != 0) ++count;
        return count;
    }
};

static_assert(sizeof(AntiDebugSnapshot) == AntiDebugSnapshot::OBSERVATION_SIZE,
              "observation layout must stay stable");
static_assert(ProviderLike<AntiDebugSnapshot>);

}  // namespace veil::providers
```

```cpp
#pragma once

#include "../config.hpp"
#include "../crypto/sha256.hpp"
#include "../detail/observe.hpp"
#include "../platform/image_file.hpp"

#include <cstddef>
#include <cstdint>

namespace veil::providers {

// 代码完整性观测。
//
// 与 SecCore 的 CRC 度量不同，这里比较磁盘映像与内存映像的摘要差值，
// 而不是内存摘要与内嵌期望值。正常环境下差值为零，因此服务端无需
// 登记任何摘要；改动内存必然让差值非零，要归零只能同时改磁盘文件，
// 那需要重新签名。
//
// region_count 不是判定值：它描述布局，篡改会让后续槽位错位，
// 派生同样分叉。
class CodeIntegrity {
public:
    static constexpr std::size_t MAX_REGIONS = 8;
    static constexpr std::size_t DIGEST_SIZE = 32;
    static constexpr std::size_t OBSERVATION_SIZE =
        sizeof(std::uint32_t) + MAX_REGIONS * DIGEST_SIZE;

    struct ObservationBuffer {
        std::uint32_t region_count;
        std::uint8_t deltas[MAX_REGIONS][DIGEST_SIZE];
    };

    static_assert(sizeof(ObservationBuffer) == OBSERVATION_SIZE,
                  "observation buffer must be tightly packed");

    // 登记当前模块的全部可执行段。磁盘侧与内存侧成对登记。
    bool register_module_sections() noexcept {
        if (!image_.open_self())
            return false;

        const std::size_t count = image_.section_count();
        for (std::size_t i = 0; i < count && region_count_ < MAX_REGIONS; ++i) {
            platform::SectionView section{};
            if (!image_.section_at(i, &section))
                continue;
            if (!section.executable || section.raw_size == 0)
                continue;
            region_sections_[region_count_++] = i;
        }
        return region_count_ > 0;
    }

    // 手工登记一对同源指针。磁盘侧与内存侧长度相同，差值由内容决定。
    // 供无法读取磁盘映像的场景使用，也供测试注入。
    bool register_region_manual(const void* disk_bytes,
                                const void* memory_bytes,
                                std::size_t size = sizeof(std::uint64_t)) noexcept {
        if (region_count_ >= MAX_REGIONS || disk_bytes == nullptr ||
            memory_bytes == nullptr || size == 0)
            return false;

        manual_regions_[region_count_] = {disk_bytes, memory_bytes, size};
        region_sections_[region_count_] = MANUAL_MARKER;
        ++region_count_;
        return true;
    }

    std::size_t region_count() const noexcept { return region_count_; }

    Observation observe() noexcept {
        buffer_ = ObservationBuffer{};
        buffer_.region_count = static_cast<std::uint32_t>(region_count_);

        for (std::size_t i = 0; i < region_count_; ++i) {
            std::uint8_t disk_digest[DIGEST_SIZE]{};
            std::uint8_t memory_digest[DIGEST_SIZE]{};

            if (region_sections_[i] == MANUAL_MARKER) {
                const ManualRegion& region = manual_regions_[i];
                Sha256::hash(region.disk_bytes, region.size, disk_digest);
                Sha256::hash(region.memory_bytes, region.size, memory_digest);
            } else {
                const std::size_t index = region_sections_[i];
                if (!image_.digest_disk(index, disk_digest))
                    continue;
                if (!image_.digest_memory(index, memory_digest))
                    continue;
            }

            for (std::size_t k = 0; k < DIGEST_SIZE; ++k)
                buffer_.deltas[i][k] =
                    static_cast<std::uint8_t>(disk_digest[k] ^ memory_digest[k]);

            detail::secure_wipe(disk_digest, sizeof(disk_digest));
            detail::secure_wipe(memory_digest, sizeof(memory_digest));
        }

        return Observation{&buffer_, OBSERVATION_SIZE};
    }

    // 测试专用：写入合成差值，验证观测链路会把它带出来。
    bool inject_delta_for_test(std::size_t index, std::uint8_t value) noexcept {
        if (index >= MAX_REGIONS)
            return false;
        synthetic_[index] = value;
        return true;
    }

private:
    static constexpr std::size_t MANUAL_MARKER = static_cast<std::size_t>(-1);

    struct ManualRegion {
        const void* disk_bytes;
        const void* memory_bytes;
        std::size_t size;
    };

    platform::ImageFile image_;
    std::size_t region_count_ = 0;
    std::size_t region_sections_[MAX_REGIONS]{};
    ManualRegion manual_regions_[MAX_REGIONS]{};
    std::uint8_t synthetic_[MAX_REGIONS]{};
    ObservationBuffer buffer_{};
};

}  // namespace veil::providers
```

```cpp
#include "../detail/observe.hpp"
#include "../platform/peb.hpp"

#include <cstddef>
#include <cstdint>

namespace veil::providers {

// 模块拓扑观测。
//
// 破解包的注入顺序是先在宿主里驻留代理 DLL，再注入目标模块。因此
// 存在非系统目录、非同目录的模块即说明环境已被污染。
//
// 输出的是计数与路径哈希累积，不是「是否可信」。路径哈希用 XOR 累积，
// 单个模块被移除或替换都会让累积值改变。
class ModuleTopology {
public:
    static constexpr std::size_t OBSERVATION_SIZE =
        sizeof(std::uint32_t) + 16;

    struct ObservationBuffer {
        std::uint32_t foreign_count;
        std::uint8_t path_hash_xor[16];
    };

    static_assert(sizeof(ObservationBuffer) == OBSERVATION_SIZE,
                  "observation buffer must be tightly packed");

    // 前缀匹配，且要求路径在可信前缀之后紧跟分隔符或结尾。
    // 任意位置子串匹配会把 c:\temp\x\microsoft.minecraftuwp_\evil.dll
    // 判为可信，这正是旧实现的问题。
    static bool is_path_trusted(const wchar_t* path,
                                std::uint32_t path_length) noexcept {
        if (path == nullptr || path_length == 0)
            return false;

        static const wchar_t* const kTrustedPrefixes[] = {
            L"c:\\windows\\system32",
            L"c:\\windows\\syswow64",
            L"c:\\windows\\winsxs",
            L"c:\\windows\\systemapps",
            L"c:\\program files\\windowsapps",
        };

        for (const wchar_t* prefix : kTrustedPrefixes) {
            std::uint32_t prefix_len = 0;
            while (prefix[prefix_len] != L'\0')
                ++prefix_len;
            if (prefix_len > path_length)
                continue;
            if (!platform::starts_with_ci(path, path_length, prefix))
                continue;

            // 前缀之后必须是分隔符或字符串结束，否则是形近目录
            if (prefix_len == path_length)
                return true;
            const wchar_t next = path[prefix_len];
            if (next == L'\\' || next == L'/')
                return true;
        }

        // 与自身模块同目录的文件视为可信。这里用「目录部分与自身一致」
        // 判定，而不是对路径做子串匹配。
        return is_same_directory_as_self(path, path_length);
    }

    static Observation observe() noexcept {
        ObservationBuffer buffer{};

        const std::size_t count = platform::module_count();
        for (std::size_t i = 0; i < count; ++i) {
            platform::ModuleInfo info{};
            if (!platform::module_at(i, &info))
                continue;
            if (is_path_trusted(info.path, info.path_length))
                continue;

            ++buffer.foreign_count;

            std::uint64_t hash = 0xCBF29CE484222325ULL;
            for (std::uint32_t k = 0; k < info.path_length; ++k)
                hash = (hash ^ static_cast<std::uint64_t>(info.path[k])) *
                       0x100000001B3ULL;

            for (std::size_t b = 0; b < 8; ++b)
                buffer.path_hash_xor[b] ^= static_cast<std::uint8_t>(hash >> (b * 8));
            for (std::size_t b = 0; b < 8; ++b)
                buffer.path_hash_xor[8 + b] ^=
                    static_cast<std::uint8_t>(hash >> ((b + 8) * 8));
        }

        static thread_local ObservationBuffer slot;
        slot = buffer;
        return Observation{&slot, OBSERVATION_SIZE};
    }

private:
    static bool is_same_directory_as_self(const wchar_t* path,
                                          std::uint32_t path_length) noexcept {
        wchar_t self[MAX_PATH];
        const DWORD written = GetModuleFileNameW(nullptr, self, MAX_PATH);
        if (written == 0 || written >= MAX_PATH)
            return false;

        std::uint32_t self_dir = written;
        while (self_dir > 0 && self[self_dir - 1] != L'\\' && self[self_dir - 1] != L'/')
            --self_dir;
        if (self_dir == 0)
            return false;

        std::uint32_t path_dir = path_length;
        while (path_dir > 0 && path[path_dir - 1] != L'\\' && path[path_dir - 1] != L'/')
            --path_dir;

        if (self_dir != path_dir)
            return false;

        for (std::uint32_t i = 0; i < self_dir; ++i) {
            if (platform::lower_ascii(self[i]) != platform::lower_ascii(path[i]))
                return false;
        }
        return true;
    }
};

static_assert(ProviderLike<ModuleTopology>);

}  // namespace veil::providers
```

```cpp
namespace veil::providers {

// 对端证书公钥指纹观测。
//
// 只返回实测值，不与任何预期值比对。服务端以自己的证书做同样的派生，
// 指纹不符时两侧派生出不同密钥，AEAD 解不开。没有可比对的布尔量。
//
// 未走 TLS 时为零值，派生结果与任何正常服务端都不一致，等价于拒绝。
struct CertificatePin {
    static constexpr std::size_t OBSERVATION_SIZE = 32;

    struct ObservationBuffer {
        std::uint8_t spki_digest[32];
    };

    static_assert(sizeof(ObservationBuffer) == OBSERVATION_SIZE,
                  "observation buffer must be tightly packed");

    static Observation from_digest(const std::uint8_t digest[32]) noexcept {
        static thread_local ObservationBuffer slot;
        slot = ObservationBuffer{};
        if (digest != nullptr) {
            for (std::size_t i = 0; i < 32; ++i)
                slot.spki_digest[i] = digest[i];
        }
        return Observation{&slot, OBSERVATION_SIZE};
    }
};

static_assert(ProviderLike<CertificatePin>);

}  // namespace veil::providers
```

```cpp
#pragma once

#include "../config.hpp"
#include "../detail/observe.hpp"

#include <cstddef>
#include <cstdint>

namespace veil::evidence {

// 按声明顺序拼接 provider 观测。
//
// 顺序与内容在编译期确定，无运行时分发、无中心注册表。消费者改动顺序
// 或增删 provider 后，服务端必须同步，否则派生分叉。
template <typename... Providers>
class Layout {
public:
    static_assert((ProviderLike<Providers> && ...),
                  "every layout member must satisfy ProviderLike");

    static constexpr std::size_t SIZE = (Providers::OBSERVATION_SIZE + ... + 0);

    static_assert(SIZE <= config::MAX_EVIDENCE_BYTES,
                  "evidence blob exceeds the configured maximum");

    static bool serialize(std::uint8_t* out, std::size_t capacity) noexcept {
        if (out == nullptr || capacity < SIZE)
            return false;

        std::size_t cursor = 0;
        const bool ok = (append<Providers>(out, cursor) && ...);
        return ok;
    }

private:
    template <typename Provider>
    static bool append(std::uint8_t* out, std::size_t& cursor) noexcept {
        const Observation observation = Provider::observe();
        if (observation.data == nullptr ||
            observation.size != Provider::OBSERVATION_SIZE)
            return false;

        const auto* bytes = static_cast<const std::uint8_t*>(observation.data);
        for (std::size_t i = 0; i < observation.size; ++i)
            out[cursor + i] = bytes[i];
        cursor += observation.size;
        return true;
    }
};

}  // namespace veil::evidence
```

```cpp
#pragma once

#include "../config.hpp"
#include "../crypto/hkdf.hpp"
#include "../detail/common.hpp"

#include <cstddef>
#include <cstdint>

namespace veil::evidence {

// 以证据为 info 派生会话密钥。
//
// 这是库内唯一的密钥派生入口。SecCore 曾并存两条派生路径，误用其一
// 即绕过整套证据链；veil 不提供不含证据的派生函数。
class Binder {
public:
    static constexpr std::size_t KEY_SIZE = 32;

    static bool derive(const void* key_material, std::size_t key_len,
                       const void* nonce, std::size_t nonce_len,
                       const void* evidence, std::size_t evidence_len,
                       std::uint8_t domain,
                       std::uint8_t out[KEY_SIZE]) noexcept {
        if (key_material == nullptr || key_len == 0)
            return false;
        if (nonce == nullptr || nonce_len == 0)
            return false;
        if (evidence == nullptr || evidence_len > config::MAX_EVIDENCE_BYTES)
            return false;
        if (out == nullptr)
            return false;

        // info = evidence ‖ domain。长度固定，服务端按同一布局复现。
        std::uint8_t info[config::MAX_EVIDENCE_BYTES + 1];
        const auto* bytes = static_cast<const std::uint8_t*>(evidence);
        for (std::size_t i = 0; i < evidence_len; ++i)
            info[i] = bytes[i];
        info[evidence_len] = domain;

        hkdf_sha256(key_material, key_len, nonce, nonce_len,
                    info, evidence_len + 1, out, KEY_SIZE);

        detail::secure_wipe(info, sizeof(info));
        return true;
    }
};

}  // namespace veil::evidence
```

```cpp
#pragma once

#include "../config.hpp"

#include <concepts>
#include <cstddef>
#include <cstdint>

namespace veil::license {

// 载荷对库不透明。库只要求长度可确定、字段可读取时效与功能位。
template <typename T>
concept PayloadLike = requires(const T& payload, std::int64_t now) {
    { sizeof(T) } -> std::convertible_to<std::size_t>;
    { payload.issued_at } -> std::convertible_to<std::uint64_t>;
    { payload.expires_at } -> std::convertible_to<std::uint64_t>;
    { payload.feature_mask } -> std::convertible_to<std::uint32_t>;
    requires std::is_trivially_copyable_v<T>;
};

// 时效折算。过期或时钟异常返回 0，调用方无需另设「是否过期」分支。
// 单一判定点正是被 patch 的目标，因此不提供独立的过期检查接口。
template <PayloadLike T>
VEIL_FORCEINLINE std::uint32_t effective_mask(const T& payload,
                                              std::int64_t now_ms) noexcept {
    if (payload.expires_at != 0) {
        const auto expires = static_cast<std::int64_t>(payload.expires_at);
        if (expires <= now_ms)
            return 0;
    }

    const auto issued = static_cast<std::int64_t>(payload.issued_at);
    if (issued > now_ms + 3600LL * 1000LL)
        return 0;

    return payload.feature_mask;
}

}  // namespace veil::license
```

`include/veil/license/envelope.hpp`：

```cpp
#pragma once

#include "../config.hpp"
#include "../crypto/hkdf.hpp"
#include "../crypto/hmac.hpp"
#include "../detail/common.hpp"
#include "payload.hpp"

#include <cstddef>
#include <cstdint>

namespace veil::license {

inline constexpr std::size_t NONCE_SIZE = 16;
inline constexpr std::size_t BINDING_SIZE = 16;
inline constexpr std::size_t SIGNATURE_SIZE = 64;
inline constexpr std::size_t MAX_MACHINE_ID = 127;
inline constexpr std::size_t MAX_BINDING_MESSAGE = 8192;
inline constexpr std::uint8_t DOMAIN_KEYSTREAM = 0x01;
inline constexpr std::uint8_t DOMAIN_BINDING = 0x02;

// 机器标识参与派生，而不是参与独立比对。
//
// 机器不符时绑定校验不同，签名自然验不过，因此不存在「机器是否匹配」
// 这样可被单点 patch 的分支。校验值由客户端本地算出，不随包传输，
// 攻击者没有可剥离的字段来退回无绑定模式。
//
// 超长标识显式失败。截断会让两台前缀相同的机器派生出同一密钥。
inline bool derive_machine_key(const void* card_key, std::size_t card_key_len,
                               const void* nonce, std::size_t nonce_len,
                               const void* machine_id, std::size_t machine_id_len,
                               std::uint8_t domain,
                               void* out, std::size_t out_len) noexcept {
    if (card_key == nullptr || card_key_len == 0)
        return false;
    if (nonce == nullptr || nonce_len == 0)
        return false;
    if (out == nullptr || out_len == 0)
        return false;
    if (machine_id_len > MAX_MACHINE_ID)
        return false;
    if (machine_id == nullptr && machine_id_len != 0)
        return false;

    std::uint8_t info[MAX_MACHINE_ID + 1];
    std::size_t info_len = 0;
    const auto* id = static_cast<const std::uint8_t*>(machine_id);
    for (std::size_t i = 0; i < machine_id_len; ++i)
        info[info_len++] = id[i];
    info[info_len++] = domain;

    hkdf_sha256(card_key, card_key_len, nonce, nonce_len,
                info, info_len, out, out_len);

    detail::secure_wipe(info, sizeof(info));
    return true;
}

// 原地异或密钥流。加密与解密是同一操作。
inline bool xor_stream(const void* card_key, std::size_t card_key_len,
                       const void* nonce,
                       const void* machine_id, std::size_t machine_id_len,
                       void* payload, std::size_t payload_len) noexcept {
    if (payload == nullptr || payload_len == 0)
        return false;

    std::uint8_t keystream_key[32];
    if (!derive_machine_key(card_key, card_key_len, nonce, NONCE_SIZE,
                            machine_id, machine_id_len,
                            DOMAIN_KEYSTREAM, keystream_key,
                            sizeof(keystream_key)))
        return false;

    std::uint8_t stream[512];
    if (payload_len > sizeof(stream)) {
        detail::secure_wipe(keystream_key, sizeof(keystream_key));
        return false;
    }

    hkdf_sha256(keystream_key, sizeof(keystream_key), nonce, NONCE_SIZE,
                nullptr, 0, stream, payload_len);

    auto* bytes = static_cast<std::uint8_t*>(payload);
    for (std::size_t i = 0; i < payload_len; ++i)
        bytes[i] ^= stream[i];

    detail::secure_wipe(keystream_key, sizeof(keystream_key));
    detail::secure_wipe(stream, sizeof(stream));
    return true;
}

// 绑定校验 = HMAC(HKDF(卡密, nonce, HWID ‖ 域), nonce ‖ 密文) 前 16 字节。
inline bool compute_binding(const void* card_key, std::size_t card_key_len,
                            const void* nonce,
                            const void* machine_id, std::size_t machine_id_len,
                            const void* cipher, std::size_t cipher_len,
                            std::uint8_t out[BINDING_SIZE]) noexcept {
    if (nonce == nullptr || cipher == nullptr || out == nullptr)
        return false;
    if (cipher_len + NONCE_SIZE > MAX_BINDING_MESSAGE)
        return false;

    std::uint8_t binding_key[32];
    if (!derive_machine_key(card_key, card_key_len, nonce, NONCE_SIZE,
                            machine_id, machine_id_len,
                            DOMAIN_BINDING, binding_key, sizeof(binding_key)))
        return false;

    std::uint8_t message[MAX_BINDING_MESSAGE];
    for (std::size_t i = 0; i < NONCE_SIZE; ++i)
        message[i] = static_cast<const std::uint8_t*>(nonce)[i];
    for (std::size_t i = 0; i < cipher_len; ++i)
        message[NONCE_SIZE + i] = static_cast<const std::uint8_t*>(cipher)[i];

    std::uint8_t digest[32];
    Hmac::compute(binding_key, sizeof(binding_key), message,
                  NONCE_SIZE + cipher_len, digest);

    for (std::size_t i = 0; i < BINDING_SIZE; ++i)
        out[i] = digest[i];

    detail::secure_wipe(binding_key, sizeof(binding_key));
    detail::secure_wipe(message, NONCE_SIZE + cipher_len);
    detail::secure_wipe(digest, sizeof(digest));
    return true;
}

}  // namespace veil::license
```

注意：`MAX_BINDING_MESSAGE` 必须在 `compute_binding` 之前声明。
把该常量定义移到文件顶部 `DOMAIN_BINDING` 之后。

```cpp
#pragma once

#include "config.hpp"
#include "evidence/binder.hpp"
#include "evidence/layout.hpp"
#include "providers/entropy.hpp"

#include <cstddef>
#include <cstdint>

namespace veil {

// 装配入口。
//
// 不在构造时做任何文件读取或加密初始化：DllMain 持有加载器锁，
// 在其中调用 CreateFile 或首次加载 bcrypt 会死锁。全部工作推迟到
// 首次派生调用。
class Guard {
public:
    Guard() noexcept = default;

    Guard(const Guard&) = delete;
    Guard& operator=(const Guard&) = delete;

    // 以布局采集的观测为 info 派生会话密钥。
    //
    // 观测不满足零值基线时，派生出的是错误密钥而非「被拒绝」。
    // 调用方拿到的密钥看似合法，故障在后续 AEAD 认证阶段才显现。
    template <typename Layout>
    bool derive_session_key(const void* card_key, std::size_t card_key_len,
                            const void* nonce, std::size_t nonce_len,
                            std::uint8_t domain,
                            std::uint8_t out[evidence::Binder::KEY_SIZE]) noexcept {
        std::uint8_t buffer[Layout::SIZE];
        if (!Layout::serialize(buffer, sizeof(buffer)))
            return false;

        const bool ok = evidence::Binder::derive(card_key, card_key_len,
                                                 nonce, nonce_len,
                                                 buffer, sizeof(buffer),
                                                 domain, out);
        detail::secure_wipe(buffer, sizeof(buffer));
        return ok;
    }

    // 采集一次观测并返回原始字节，供诊断工具与跨实现夹具使用。
    // 业务路径不得以此结果做任何判定。
    template <typename Layout>
    bool capture(std::uint8_t* out, std::size_t capacity) noexcept {
        return Layout::serialize(out, capacity);
    }
};

}  // namespace veil
```
